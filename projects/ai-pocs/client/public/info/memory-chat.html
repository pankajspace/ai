Concept#

🧠 Conversational Memory enables multi-turn dialogue by injecting prior user and assistant turns into a LangChain MessagesPlaceholder before each new LLM call.

LLMs are inherently stateless—they do not remember previous HTTP requests unless the conversation history is re-sent in the prompt. This service converts the client's history list into LangChain HumanMessage and AIMessage objects, splices them into the prompt via MessagesPlaceholder(variable_name="history"), and returns the updated transcript.

2

Theory & Concepts#

Core Concepts:

  • Stateless Inference vs. Stateful UX: Conversational memory is an application-layer pattern where prior turns are replayed in the context window on every turn.
  • MessagesPlaceholder: Dynamically expands a list of typed message objects (SystemMessage, HumanMessage, AIMessage) at a specific position in a ChatPromptTemplate.
  • Client-Carried State: Returning the updated history array in the response lets the React frontend persist or reset the session without server-side sticky sessions.

Request flow#

🧑 Browser MemoryChatCard.tsx sends POST /api/memory-chat with the new message and existing history array.

→
History Reconstruction (04_langchain_memory_chat.py) chat_with_memory() converts stored turns into HumanMessage and AIMessage instances.
→

→
🧑 Browser The new user message and assistant reply are appended to updated_history and returned in MemoryChatResponse.

server/services/04_langchain_memory_chat.py

LangChain Conversational Memory Chat Service.

"""LangChain Conversational Memory Chat Service.

This service demonstrates conversational memory using LangChain LCEL (LangChain Expression Language):
1. Loads API credentials and model configuration from environment variables (.env).
2. Maintains conversation context across multiple turns using ChatPromptTemplate with MessagesPlaceholder.
3. Uses Pydantic models (ChatMessage, MemoryChatRequest, MemoryChatResponse) for structured serialization.
4. Dynamically appends user questions and assistant answers to preserve context across turns.

Run directly:
    python projects/ai-pocs/server/services/04_langchain_memory_chat.py
"""

import os
import sys
from typing import List, Optional
from dotenv import find_dotenv, load_dotenv
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
from langchain_openai import ChatOpenAI
from pydantic import BaseModel, Field

# ---------------------------------------------------------------------------
# Configuration & Environment Setup
# ---------------------------------------------------------------------------
load_dotenv(find_dotenv())


# ---------------------------------------------------------------------------
# Request & Response Schemas (Pydantic)
# ---------------------------------------------------------------------------
class ChatMessage(BaseModel):
    """Single chat turn in conversation history."""

    role: str = Field(
        default="user",
        description="Author role: 'user' or 'assistant'",
    )
    content: str = Field(
        default="",
        description="Text content of the message",
    )


class MemoryChatRequest(BaseModel):
    """Input payload for memory chat queries."""

    message: str = Field(
        default="My favorite color is navy blue.",
        description="Current user prompt or question",
    )
    history: List[ChatMessage] = Field(
        default_factory=list,
        description="Previous conversation turns preserving message context",
    )


class MemoryChatResponse(BaseModel):
    """Output payload containing assistant response and updated history."""

    message: str
    response: str
    history: List[ChatMessage]


# ---------------------------------------------------------------------------
# Core Service Logic
# ---------------------------------------------------------------------------
def chat_with_memory(
    question: str = "My favorite color is navy blue.",
    history: Optional[List[ChatMessage]] = None,
) -> tuple[str, List[ChatMessage]]:
    """Execute a conversational chat turn with memory retention.

    Args:
        question: User message or query.
        history: Previous chat messages to provide context.

    Returns:
        Tuple of (assistant_response, updated_chat_history).
    """
    # ① Default to an empty list and format history into (role, content) tuples for MessagesPlaceholder
    history_list = history or []
    lc_history = [(msg.role, msg.content) for msg in history_list]

    # ② Retrieve model parameters and initialize the ChatOpenAI client
    model_name = os.getenv("OPENAI_MODEL", "gpt-4o-mini")
    base_url = os.getenv("OPENAI_BASE_URL")
    api_key = os.getenv("OPENAI_API_KEY")

    model = ChatOpenAI(
        model=model_name,
        api_key=api_key,
        base_url=base_url if base_url else None,
        timeout=15.0,
    )

    # ③ Build prompt template injecting system instructions, chat history, and current question
    prompt = ChatPromptTemplate.from_messages([
        (
            "system",
            "You are a friendly assistant. Use the conversation history to stay consistent.",
        ),
        MessagesPlaceholder("history"),  # Dynamically receives lc_history tuples
        ("human", "{question}"),
    ])

    # ④ Assemble LCEL pipeline (Prompt | Model) and invoke with history and question
    chain = prompt | model
    result = chain.invoke({"history": lc_history, "question": question})
    answer = str(result.content) if hasattr(result, "content") else str(result)

    # ⑤ Build updated history list:
    # 1. *history_list: Unpacks all existing ChatMessage items into the new list (non-mutating)
    # 2. Appends the current user query and model answer to preserve context for the next turn
    updated_history = [
        *history_list,
        ChatMessage(role="user", content=question),
        ChatMessage(role="assistant", content=answer),
    ]

    return answer, updated_history




# ---------------------------------------------------------------------------
# Standalone CLI Verification
# ---------------------------------------------------------------------------
if __name__ == "__main__":
    print("--- Conversational Memory Chat Service ---")
    if sys.stdin.isatty():
        print("Chat with the bot (type 'quit' or 'exit' to stop).")
        session_history: List[ChatMessage] = []
        while True:
            try:
                user_query = input("You: ")
            except (EOFError, KeyboardInterrupt):
                print()
                break
            if user_query.strip().lower() in {"quit", "exit"}:
                break
            if not user_query.strip():
                continue

            bot_reply, session_history = chat_with_memory(user_query, session_history)
            print("Bot:", bot_reply)
            print(f"   (history now has {len(session_history)} messages)")
    else:
        print("\nTurn 1: Telling bot user's preference...")
        turn1_reply, h1 = chat_with_memory("Hi! My favorite color is navy blue.")
        print("Bot:", turn1_reply)

        print("\nTurn 2: Asking bot to recall user's preference...")
        turn2_reply, h2 = chat_with_memory("What is my favorite color?", history=h1)
        print("Bot:", turn2_reply)
        print(f"   (history preserved {len(h2)} messages)")

FastAPI route handler for POST /api/memory-chat (memory_chat_service) in server/main.py.

from dotenv import find_dotenv, load_dotenv
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware

from services import (
    ChatMessage,
    MemoryChatRequest,
    MemoryChatResponse,
    chat_with_memory,
)

# Load environment variables
load_dotenv(find_dotenv())

app = FastAPI(title="AI POCs API")

# Enable CORS for frontend requests
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)


@app.post("/api/memory-chat", response_model=MemoryChatResponse)
def memory_chat_service(req: MemoryChatRequest):
    # ① Validate user message and pass conversation history to memory chat service
    message_text = req.message.strip() if req.message else ""
    if not message_text:
        raise HTTPException(status_code=400, detail="Message cannot be empty")
    try:
        reply, updated_history = chat_with_memory(message_text, req.history)
        return MemoryChatResponse(
            message=message_text,
            response=reply,
            history=updated_history,
        )
    except Exception as exc:
        raise HTTPException(status_code=500, detail=str(exc))
