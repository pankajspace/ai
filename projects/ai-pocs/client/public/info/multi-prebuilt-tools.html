Unit 1

How It Works#

The idea behind the demo, the request it sends and the function calls that answer it.

1

Concept#

Give the agent a task like "fetch stock data from this URL, average the closing prices and save the result to stock_summary.txt". It calls http_request to get the JSON, calculator for the average, and file_write to save the summary.

file_write normally asks a human to confirm before writing. Setting BYPASS_TOOL_CONSENT=true skips that prompt so the pipeline can run unattended — and means the agent really writes files on the server, so use it with care.

2

Theory & Concepts#

  • Tool chaining: each tool's output becomes the next tool's input.
  • Tool consent: sensitive prebuilt tools ask for confirmation unless BYPASS_TOOL_CONSENT is set; this variable affects every tool in the process.
  • Written files: the card lists each file_write call's path and content, read from the agent's toolUse blocks.
3

Request flow#

🧑 Browser MultiPrebuiltToolsCard.tsx sends POST /api/multi-prebuilt-tools with { task }.
→
FastAPI route (main.py) multi_prebuilt_tools_service(req) calls run_multi_prebuilt_tools_agent(task).
→
Agent (18_multi_prebuilt_tools.py) Bedrock calls http_request → calculator → file_write.
→
🧑 Browser Shows the reply, the tool sequence and the files written.

Multi-Prebuilt Tools Agent Service.

"""Multi-Prebuilt Tools Agent Service.

Demonstrates chaining multiple prebuilt tools and unattended consent:
1. Prebuilt tools from `strands_tools`: `http_request`, `calculator`, and `file_write`.
2. Sets `BYPASS_TOOL_CONSENT="true"` so sensitive tools (`file_write`) run unattended without prompts.
3. Handles complex multi-step pipelines: Fetch external API data -> Compute statistics -> Save artifact.

Run directly:
    python projects/ai-pocs/server/services/18_multi_prebuilt_tools.py
"""

import os
from typing import Dict, List

from dotenv import find_dotenv, load_dotenv
from pydantic import BaseModel, Field
from strands import Agent
from strands.models.bedrock import BedrockModel
from strands_tools import calculator, file_write, http_request

# ① Load AWS credentials and Bedrock configuration from .env
load_dotenv(find_dotenv())


class MultiPrebuiltToolsRequest(BaseModel):
    """Input payload for Multi Prebuilt Tools Agent."""

    task: str = Field(
        default=(
            "Fetch stock data from https://query1.finance.yahoo.com/v8/finance/chart/AAPL?interval=1d&range=5d, "
            "extract the latest closing prices, calculate the average price over the period, "
            "and save the results to stock_summary.txt"
        ),
        description="Multi-step data pipeline task combining HTTP, calculation, and file writing",
    )


class MultiPrebuiltToolsResponse(BaseModel):
    """Output payload from Multi Prebuilt Tools Agent."""

    task: str
    response: str
    tool_sequence: List[str]
    written_files: Dict[str, str]
    provider: str


# ② Resolve the target Bedrock model identifier
model_id = os.getenv("BEDROCK_MODEL_ID", "amazon.nova-lite-v1:0")


# ③ Disable interactive tool consent prompts so file_write runs unattended
os.environ["BYPASS_TOOL_CONSENT"] = os.getenv("BYPASS_TOOL_CONSENT", "true")


def run_multi_prebuilt_tools_agent(
    task: str = MultiPrebuiltToolsRequest.model_fields["task"].default,
) -> dict:
    """Run the Multi Prebuilt Tools Agent."""
    # ④ Initialize the Strands Agent with http_request, calculator, and file_write
    agent = Agent(
        model=BedrockModel(model_id=model_id),
        tools=[http_request, calculator, file_write],
        system_prompt="You help with data analysis tasks.",
    )

    # ⑤ Execute the multi-step pipeline: fetch HTTP data -> compute average -> save artifact file
    response = agent(task)

    # ⑥ Collect the tool calls in order, and the files the agent asked file_write to save
    tool_uses = [
        block["toolUse"]
        for message in agent.messages
        for block in message["content"]
        if "toolUse" in block
    ]
    written_files = {t["input"]["path"]: t["input"]["content"] for t in tool_uses if t["name"] == "file_write"}

    return {
        "task": task,
        "response": str(response),
        "tool_sequence": [f"{t['name']}({t['input']})" for t in tool_uses],
        "written_files": written_files,
        "provider": f"Strands Multi-Tool + Bedrock ({model_id})",
    }


if __name__ == "__main__":
    out = run_multi_prebuilt_tools_agent()
    print("Sequence:", out["tool_sequence"])
    print("Files:", list(out["written_files"]))
    print("Response:", out["response"])

server/main.py

FastAPI route handler for POST /api/multi-prebuilt-tools (multi_prebuilt_tools_service) in server/main.py.

from dotenv import find_dotenv, load_dotenv
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware

from services import (
    MultiPrebuiltToolsRequest,
    MultiPrebuiltToolsResponse,
    run_multi_prebuilt_tools_agent,
)

# Load environment variables
load_dotenv(find_dotenv())

app = FastAPI(title="AI POCs API")

# Enable CORS for frontend requests
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)


@app.post("/api/multi-prebuilt-tools", response_model=MultiPrebuiltToolsResponse)
def multi_prebuilt_tools_service(req: MultiPrebuiltToolsRequest):
    # ① Normalize task and execute multi-prebuilt tools pipeline agent
    t = req.task.strip() if req.task else "Fetch crypto data from https://api.coindesk.com/v1/bpi/currentprice.json, extract USD rate, and write a summary to crypto_summary.txt"
    try:
        res = run_multi_prebuilt_tools_agent(t)
        return MultiPrebuiltToolsResponse(**res)
    except Exception as exc:
        raise HTTPException(status_code=500, detail=str(exc))
