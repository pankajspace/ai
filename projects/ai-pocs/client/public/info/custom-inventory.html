Concept#

📦 Custom Inventory Tool demonstrates explicit TOOL_SPEC schema contracts with regex pattern constraints (^PROD-[0-9]{3}$) for warehouse SKU lookups.

While docstring inference works for simple tools, production enterprise tools often require strict JSON Schema constraints like regex patterns, enums, and custom descriptions. Defining an explicit TOOL_SPEC dictionary guarantees that the LLM formats product IDs as PROD-123 before calling check_inventory(product_id).

2

Theory & Concepts#

Core Concepts:

  • Explicit TOOL_SPEC Contracts: Separates the external JSON Schema contract from the internal Python implementation.
  • Regex Parameter Validation: Including "pattern": "^PROD-[0-9]{3}$" in the parameter schema guides the model's token generation to valid SKU identifiers.

Custom Inventory Tool Agent Service.

"""Custom Inventory Tool Agent Service.

Demonstrates building domain-specific custom tools:
1. Simulates an internal proprietary system or inventory database.
2. Custom tool `check_inventory(product_id: str)` with strict typing & docstrings.
3. Natural language query resolution against mock product inventory.

Run directly:
    python projects/ai-pocs/server/services/16_custom_tool_inventory.py
"""

import json
import os
from typing import List
from dotenv import find_dotenv, load_dotenv
from openai import OpenAI
from pydantic import BaseModel, Field

load_dotenv(find_dotenv())


class CustomInventoryRequest(BaseModel):
    """Input payload for Custom Inventory Agent."""

    question: str = Field(
        default="Is PROD-123 in stock?",
        description="Natural language question about product availability",
    )


class CustomInventoryResponse(BaseModel):
    """Output payload from Custom Inventory Agent."""

    question: str
    response: str
    tool_calls: List[str]
    provider: str


INVENTORY = {
    "PROD-123": 15,
    "PROD-456": 0,
    "PROD-789": 8,
    "PROD-999": 0,
}


def check_inventory(product_id: str) -> str:
    """Check if a product is in stock."""
    # ① Normalize the product ID and look up quantity in INVENTORY
    pid = product_id.upper().strip()
    qty = INVENTORY.get(pid, 0)
    # ② Return distinct status messages for unknown SKU, in-stock, or out-of-stock
    if pid not in INVENTORY:
        return f"Product {pid} not found in inventory catalog."
    if qty > 0:
        return f"Product {pid} is in stock. We have {qty} units available."
    return f"Product {pid} is currently out of stock."


def run_custom_inventory_agent(question: str = "Is PROD-123 in stock?") -> dict:
    """Run the Custom Inventory Agent."""
    # ① Load AWS credentials and initialize the tool call log
    aws_key = os.getenv("AWS_ACCESS_KEY_ID")
    aws_secret = os.getenv("AWS_SECRET_ACCESS_KEY")
    bedrock_model_id = os.getenv("BEDROCK_MODEL_ID", "amazon.nova-lite-v1:0")

    tool_calls_recorded = []

    # ② Run Strands Agent on Bedrock with the custom check_inv tool if configured
    if aws_key and aws_secret:
        try:
            from strands import Agent, tool
            from strands.models.bedrock import BedrockModel

            @tool
            def check_inv(product_id: str) -> str:
                """Check if a product is in stock in the inventory."""
                res = check_inventory(product_id)
                tool_calls_recorded.append(f"check_inventory('{product_id}') -> {res}")
                return res

            agent = Agent(model=BedrockModel(model_id=bedrock_model_id), tools=[check_inv])
            reply = agent(question)
            return {
                "question": question,
                "response": str(reply),
                "tool_calls": tool_calls_recorded,
                "provider": f"Strands + Bedrock ({bedrock_model_id})",
            }
        except Exception as exc:
            print(f"Strands Bedrock inventory failed ({exc}), falling back to OpenAI")

    # ③ Configure OpenAI client and custom check_inventory tool schema
    client = OpenAI(
        api_key=os.getenv("OPENAI_API_KEY"),
        base_url=os.getenv("OPENAI_BASE_URL", "https://api.openai.com/v1"),
    )

    tools_schema = [
        {
            "type": "function",
            "function": {
                "name": "check_inventory",
                "description": "Check if a product ID (e.g. 'PROD-123', 'PROD-456', 'PROD-789') is currently in stock.",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "product_id": {"type": "string", "description": "The product ID to look up (e.g. PROD-123)"}
                    },
                    "required": ["product_id"],
                },
            },
        }
    ]

    messages = [
        {
            "role": "system",
            "content": "You are an inventory assistant. Always check product availability using the check_inventory tool.",
        },
        {"role": "user", "content": question},
    ]

    # ④ Execute the tool-calling loop to resolve SKU stock status
    for _ in range(5):
        resp = client.chat.completions.create(
            model=os.getenv("OPENAI_MODEL", "gpt-4o-mini"),
            messages=messages,
            tools=tools_schema,
            tool_choice="auto",
        )
        msg = resp.choices[0].message
        messages.append(msg)

        if not msg.tool_calls:
            return {
                "question": question,
                "response": msg.content or "",
                "tool_calls": tool_calls_recorded,
                "provider": f"OpenAI {os.getenv('OPENAI_MODEL', 'gpt-4o-mini')} (Custom Tool)",
            }

        for tc in msg.tool_calls:
            args = json.loads(tc.function.arguments or "{}")
            pid = args.get("product_id", "PROD-123")
            out = check_inventory(pid)
            tool_calls_recorded.append(f"check_inventory('{pid}') -> {out}")
            messages.append({"role": "tool", "tool_call_id": tc.id, "content": out})

    return {
        "question": question,
        "response": messages[-1].get("content", ""),
        "tool_calls": tool_calls_recorded,
        "provider": f"OpenAI {os.getenv('OPENAI_MODEL', 'gpt-4o-mini')}",
    }


if __name__ == "__main__":
    res = run_custom_inventory_agent()
    print("Tool calls:", res["tool_calls"])
    print("Response:", res["response"])

server/main.py

FastAPI route handler for POST /api/custom-inventory (custom_inventory_service) in server/main.py.

from dotenv import find_dotenv, load_dotenv
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware

from services import (
    CustomInventoryRequest,
    CustomInventoryResponse,
    run_custom_inventory_agent,
)

# Load environment variables
load_dotenv(find_dotenv())

app = FastAPI(title="AI POCs API")

# Enable CORS for frontend requests
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)


@app.post("/api/custom-inventory", response_model=CustomInventoryResponse)
def custom_inventory_service(req: CustomInventoryRequest):
    # ① Normalize question and query custom inventory tool agent
    q = req.question.strip() if req.question else "Is PROD-123 in stock?"
    try:
        res = run_custom_inventory_agent(q)
        return CustomInventoryResponse(**res)
    except Exception as exc:
        raise HTTPException(status_code=500, detail=str(exc))
