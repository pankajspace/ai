Unit 1

How It Works#

The idea behind the demo, the request it sends and the function calls that answer it.

1

Concept#

Every case runs the real agent from 30_mcp_agent.py and applies two assertions: tool routing (did it call the expected tool, or no tool for "What is 2 + 2?") and ground truth (does the answer contain e.g. "287" or "164").

Re-running the suite after changing a prompt, model or tool description shows quickly whether the agent got better or worse.

2

Theory & Concepts#

  • Evals: automated checks for LLM behaviour, run like a test suite.
  • Routing assertion: checks the agent's decisions, not just its words.
  • Ground-truth assertion: required facts must appear in the final answer.
  • Negative case: a question that needs no tool catches over-eager tool use.
3

Request flow#

🧑 Browser McpEvalsCard.tsx sends POST /api/mcp-evals.
→
FastAPI route (main.py) mcp_evals_service() runs asyncio.run(run_mcp_evals_suite()).
→

MCP Agent Evaluation Suite Service.

"""MCP Agent Evaluation Suite Service.

Runs unit-test style evaluation assertions against the MCP-enabled cricket agent (service 30):
1. Tool routing assertion: verifies the LLM chooses the expected MCP tool (`get_live_score`, `get_player_stats`, or none).
2. Ground-truth verification: checks whether required factual tokens appear in the agent's final answer.
3. Suite scoring: aggregates pass/fail counts and overall accuracy percentage across all test cases.

Run directly:
    python projects/ai-pocs/server/services/31_mcp_evals.py
"""

import asyncio
import importlib
import time
from typing import List, Optional

from pydantic import BaseModel

# ① Reuse the MCP agent loop from 30_mcp_agent.py (importlib, because the file name starts with a digit)
mcp_agent = importlib.import_module(f"{__package__}.30_mcp_agent" if __package__ else "30_mcp_agent")


class TestCaseEvalResult(BaseModel):
    case_name: str
    prompt: str
    expected_tool: Optional[str]
    actual_tool: Optional[str]
    tool_routing_passed: bool
    expected_keywords: List[str]
    answer_verified: bool
    overall_passed: bool
    agent_answer: str
    latency_seconds: float


class McpEvalsResponse(BaseModel):
    """Output payload from MCP Evals Suite."""

    total_cases: int
    passed_cases: int
    failed_cases: int
    accuracy_percentage: float
    results: List[TestCaseEvalResult]
    model: str


# ② Evaluation test suite: question, expected tool (None = no tool) and facts the answer must contain
TEST_CASES = [
    {"q": "What's the live score of India vs West Indies?", "tool": "get_live_score", "must_contain": ["287"]},
    {"q": "How many runs does West Indies need to win?", "tool": "get_live_score", "must_contain": ["90"]},
    {"q": "How many runs has Shubman Gill scored this series?", "tool": "get_player_stats", "must_contain": ["164"]},
    {"q": "Shai Hope's highest score in the series?", "tool": "get_player_stats", "must_contain": ["78"]},
    {"q": "What is 2 + 2?", "tool": None, "must_contain": ["4"]},
]


async def run_case(i: int, case: dict) -> dict:
    """Run the agent on one test case and apply both assertions."""
    start = time.perf_counter()
    result = await mcp_agent.run_agent(case["q"])
    tools_used = [step["tool"] for step in result["trace"]]
    answer = result["answer"] or ""

    # ③ Assertion 1 (Tool Routing): the agent picked the expected tool (or no tool at all)
    tool_ok = (case["tool"] in tools_used) if case["tool"] else (tools_used == [])
    # ④ Assertion 2 (Ground Truth): the final answer contains every required fact
    answer_ok = all(word.lower() in answer.lower() for word in case["must_contain"])

    return {
        "case_name": f"Test {i}",
        "prompt": case["q"],
        "expected_tool": case["tool"],
        "actual_tool": ", ".join(tools_used) or None,
        "tool_routing_passed": tool_ok,
        "expected_keywords": case["must_contain"],
        "answer_verified": answer_ok,
        "overall_passed": tool_ok and answer_ok,
        "agent_answer": answer,
        "latency_seconds": round(time.perf_counter() - start, 2),
    }


async def run_mcp_evals_suite() -> dict:
    """Execute the complete MCP evaluation suite."""
    # ⑤ Run every test case in turn
    results = [await run_case(i, case) for i, case in enumerate(TEST_CASES, 1)]

    # ⑥ Compute the final suite score
    passed = sum(r["overall_passed"] for r in results)
    return {
        "total_cases": len(results),
        "passed_cases": passed,
        "failed_cases": len(results) - passed,
        "accuracy_percentage": round(100 * passed / len(results), 1),
        "results": results,
        "model": mcp_agent.MODEL,
    }


if __name__ == "__main__":
    out = asyncio.run(run_mcp_evals_suite())
    for r in out["results"]:
        print(f"{r['case_name']}: {'PASS' if r['overall_passed'] else 'FAIL'} | {r['prompt']} | tools: {r['actual_tool']}")
    print(f"==== SCORE: {out['passed_cases']}/{out['total_cases']} passed ({out['accuracy_percentage']}%) ====")

MCP-Enabled LLM Agent Service.

"""MCP-Enabled LLM Agent Service.

Connects an LLM to the FastMCP cricket server:
1. Starts `cricket_server.py` over stdio and discovers its tools with `tools/list`.
2. Converts the MCP tool descriptors into the OpenAI function-calling format.
3. Runs the tool-calling loop: the LLM asks for a tool, the client runs it with `tools/call`,
   and the tool output is fed back until the LLM gives a final answer.

Run directly:
    python projects/ai-pocs/server/services/30_mcp_agent.py
"""

import asyncio
import json
import os
import re
import sys
from pathlib import Path
from typing import Any, Dict, Optional

from dotenv import find_dotenv, load_dotenv
from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import stdio_client
from openai import OpenAI
from pydantic import BaseModel, Field

load_dotenv(find_dotenv())


class McpAgentRequest(BaseModel):
    """Input payload for MCP Agent."""

    question: str = Field(
        default="What is the score in India vs West Indies?",
        description="Natural language question about cricket scores or player stats",
    )


class McpAgentResponse(BaseModel):
    """Output payload from MCP Agent."""

    question: str
    response: str
    tool_called: Optional[str]
    tool_arguments: Optional[Dict[str, Any]]
    tool_result: Optional[Dict[str, Any]]
    provider: str


# ① Model settings (Groq's OpenAI-compatible endpoint) and the MCP server to spawn over stdio
MODEL = os.getenv("GROQ_MODEL", "openai/gpt-oss-20b")
GROQ_BASE_URL = os.getenv("GROQ_BASE_URL", "https://api.groq.com/openai/v1")
server_script = str(Path(__file__).with_name("cricket_server.py"))
server = StdioServerParameters(command=sys.executable, args=[server_script])


def to_openai_format(mcp_tools):
    """MCP tool list -> format expected by OpenAI-compatible API."""
    return [
        {
            "type": "function",
            "function": {
                "name": t.name,
                "description": t.description or "",
                "parameters": t.inputSchema,
            },
        }
        for t in mcp_tools
    ]


def call_llm(llm, kwargs, retries=2):
    """Call the model, retrying when it tries to call a tool we never gave it."""
    last = None
    for _ in range(retries + 1):
        try:
            return llm.chat.completions.create(**kwargs).choices[0].message, None
        except Exception as e:
            last = str(e)
            if "tool_use_failed" not in last:
                return None, f"API ERROR: {last}"
    m = re.search(r'"name":\s*"([^"]+)"', last or "")
    tool = m.group(1) if m else "a tool"
    return None, f"MODEL TRIED TO CALL A TOOL WE NEVER GAVE IT ({tool}). Try running again."


async def run_agent(question: str) -> dict:
    """Run the MCP agent loop for a single user question; returns the answer and a tool trace."""
    # ② Initialize the Groq OpenAI-compatible client
    api_key = os.getenv("GROQ_API_KEY")
    if not api_key:
        raise ValueError("GROQ_API_KEY is not set. Free key: https://console.groq.com/keys")
    llm = OpenAI(api_key=api_key, base_url=GROQ_BASE_URL)
    trace = []

    messages = [
        {
            "role": "system",
            "content": (
                "You are a cricket assistant. Use ONLY the functions provided to get live data. "
                "Do not use any browser or code tools. Be brief."
            ),
        },
        {"role": "user", "content": question},
    ]

    # ③ Connect to the FastMCP cricket server over stdio and discover its tools
    async with stdio_client(server) as (read, write):
        async with ClientSession(read, write) as session:
            await session.initialize()
            tools = to_openai_format((await session.list_tools()).tools)

            # ④ Run the multi-step tool-calling loop (up to 5 turns)
            for _ in range(5):
                reply, error = call_llm(llm, {"model": MODEL, "messages": messages, "tools": tools, "temperature": 0})
                if error:
                    return {"answer": f"({error})", "trace": trace}

                # No tool requested -> this is the final answer
                if not reply.tool_calls:
                    return {"answer": reply.content or "", "trace": trace}

                # ⑤ Run each requested tool on the MCP server and feed the output back to the LLM
                messages.append(
                    {
                        "role": "assistant",
                        "content": reply.content or "",
                        "tool_calls": [
                            {
                                "id": c.id,
                                "type": "function",
                                "function": {"name": c.function.name, "arguments": c.function.arguments},
                            }
                            for c in reply.tool_calls
                        ],
                    }
                )
                for call in reply.tool_calls:
                    args = json.loads(call.function.arguments)
                    result = await session.call_tool(call.function.name, args)
                    text = result.content[0].text if result.content else "{}"
                    trace.append({
                        "tool": call.function.name,
                        "args": args,
                        "result": {"error": text} if result.isError else json.loads(text),
                    })
                    messages.append({"role": "tool", "tool_call_id": call.id, "content": text})

    return {"answer": "(stopped: too many steps)", "trace": trace}


async def run_mcp_agent(question: str = "What is the score in India vs West Indies?") -> dict:
    """Run the MCP-enabled agent and report the first tool it used."""
    out = await run_agent(question)
    first = out["trace"][0] if out["trace"] else None
    return {
        "question": question,
        "response": out["answer"],
        "tool_called": first["tool"] if first else None,
        "tool_arguments": first["args"] if first else None,
        "tool_result": first["result"] if first else None,
        "provider": f"Groq ({MODEL}) + FastMCP",
    }


if __name__ == "__main__":
    out = asyncio.run(run_mcp_agent())
    print("Tool Called:", out["tool_called"], out["tool_arguments"])
    print("Response:", out["response"])

Cricket FastMCP Server (`cricket_server.py`).

"""Cricket FastMCP Server (`cricket_server.py`).

Provides Model Context Protocol (MCP) tool endpoints for live cricket matches and player statistics:
1. `get_live_score(teams)`: Fetches the live scorecard, target, and match situation.
2. `get_player_stats(player_name)`: Fetches series batting statistics and highest scores.
3. Communicates over standard input/output (`stdio`) transport so MCP clients can spawn it as a subprocess.

Note: The match and player data below is mock/demo data for classroom exercises (no external API key needed).

Run directly (normally the MCP clients 29-31 start it for you as a subprocess):
    python projects/ai-pocs/server/services/cricket_server.py
"""

import sys
from pydantic import Field
from mcp.server.fastmcp import FastMCP

# ---------------------------------------------------------------------------
# FastMCP Server Initialization
# ---------------------------------------------------------------------------
mcp = FastMCP("Cricket-Score-Server")

# ---------------------------------------------------------------------------
# Mock Data Store (Simulated Third-Party Sports API Data)
# ---------------------------------------------------------------------------
LIVE_MATCHES = {
    "india vs west indies": {
        "series": "India vs West Indies ODI Series, Sep-Oct 2026 (DEMO DATA)",
        "match": "2nd ODI",
        "status": "LIVE",
        "india": "287/6 (50 overs)",
        "west_indies": "198/5 (38.2 overs)",
        "target": 288,
        "required": "90 runs from 70 balls",
        "series_score": "India lead 1-0",
    }
}

PLAYER_STATS = {
    "shubman gill": {"team": "India", "series_runs": 164, "innings": 3, "highest": 150},
    "shai hope": {"team": "West Indies", "series_runs": 131, "innings": 2, "highest": 78},
}


# ---------------------------------------------------------------------------
# FastMCP Tool Definitions
# ---------------------------------------------------------------------------
@mcp.tool(description="Get the LIVE score of a cricket match. Use this for any question about the current score, who is winning, or runs required.")
def get_live_score(teams: str = Field(description="The two teams, e.g. 'India vs West Indies'")) -> dict:
    """Get the LIVE score of a cricket match."""
    # ① Log diagnostic trace to stderr (stdout is reserved strictly for MCP JSON-RPC frames)
    print(f"[server] get_live_score called with teams={teams!r}", file=sys.stderr)  # logs go to stderr, NEVER stdout
    # ② Normalize team names and look up the live match scorecard
    key = teams.strip().lower().replace("windies", "west indies")
    if key not in LIVE_MATCHES:
        return {"error": f"No live match found for '{teams}'. Try 'India vs West Indies'."}
    return LIVE_MATCHES[key]


@mcp.tool(description="Get a player's batting stats for the current series. Use this for questions about a specific player's runs.")
def get_player_stats(player_name: str = Field(description="Full player name, e.g. 'Shubman Gill'")) -> dict:
    """Get a player's batting stats for the current series."""
    # ① Log diagnostic trace to stderr
    print(f"[server] get_player_stats called with player_name={player_name!r}", file=sys.stderr)
    # ② Normalize player name and return series batting statistics
    key = player_name.strip().lower()
    if key not in PLAYER_STATS:
        return {"error": f"No stats found for '{player_name}'."}
    return PLAYER_STATS[key]


# ---------------------------------------------------------------------------
# Entry Point (Stdio Transport)
# ---------------------------------------------------------------------------
if __name__ == "__main__":
    mcp.run(transport="stdio")

FastAPI route handler for POST /api/mcp-evals (mcp_evals_service) in server/main.py.

import asyncio

from dotenv import find_dotenv, load_dotenv
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware

from services import (
    McpEvalsResponse,
    run_mcp_evals_suite,
)

# Load environment variables
load_dotenv(find_dotenv())

app = FastAPI(title="AI POCs API")

# Enable CORS for frontend requests
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)


@app.post("/api/mcp-evals", response_model=McpEvalsResponse)
def mcp_evals_service():
    # ① Execute MCP agent evaluation suite and return assertion metrics
    try:
        res = asyncio.run(run_mcp_evals_suite())
        return McpEvalsResponse(**res)
    except Exception as exc:
        raise HTTPException(status_code=500, detail=str(exc))
