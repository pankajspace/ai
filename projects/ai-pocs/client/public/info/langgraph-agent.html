How It Works#

The idea behind the demo, the request it sends and the function calls that answer it.

Concept#

🔄 LangGraph ReAct Agent builds a graph-based ReAct (Reason + Act) agent that calls a custom say_hello(name) tool for each person mentioned in the prompt.

When asked to "Please greet Alice and Bob", the ReAct state machine loops between the reasoning node (LLM) and the action node (say_hello tool) until every person has been greeted, recording each tool call in the state graph's message list.

2

Theory & Concepts#

Core Concepts:

  • ReAct State Graph: Models agent execution as a directed cyclic graph between an agent node (LLM decision) and a tools node (function execution).
  • Multi-Entity Tool Fan-Out: The LLM extracts multiple named entities (Alice, Bob) from a single prompt and invokes say_hello for each one.
3

Request flow#

🧑 Browser LangGraphAgentCard.tsx sends POST /api/langgraph-agent with {"prompt": "Please greet Alice and Bob."}.
→
FastAPI Route (server/main.py) langgraph_agent_service(req) calls run_langgraph_agent(prompt_text).
→
ReAct Graph Execution (12_hello_world_langgraph.py) Constructs the LangGraph ReAct agent (with OpenAI tool-loop fallback) bound to say_hello(name).
→
Tool Execution & Synthesis Executes say_hello("Alice") and say_hello("Bob"), then synthesizes the final greeting.
→
🧑 Browser Returns LangGraphAgentResponse with response, tools_called, and framework.

server/services/12_hello_world_langgraph.py

LangGraph ReAct Hello World Agent Service.

"""LangGraph ReAct Hello World Agent Service.

Demonstrates the LangGraph tool-calling ReAct agentic loop:
1. Defines a tool `greet(name: str)`.
2. Passes prompt to LangGraph ReAct agent.
3. Observes tool execution steps (calling greet for each person).
4. Synthesizes final response.

Run directly:
    python projects/ai-pocs/server/services/12_hello_world_langgraph.py
"""

import json
import os
from typing import List
from dotenv import find_dotenv, load_dotenv
from openai import OpenAI
from pydantic import BaseModel, Field

load_dotenv(find_dotenv())


class LangGraphAgentRequest(BaseModel):
    """Input payload for LangGraph Hello World Agent."""

    prompt: str = Field(
        default="Please greet Alice and Bob.",
        description="User request for greeting individuals",
    )


class LangGraphAgentResponse(BaseModel):
    """Output payload from LangGraph Hello World Agent."""

    prompt: str
    response: str
    tool_calls: List[str]
    provider: str


def greet_tool(name: str) -> str:
    """Greet someone by name."""
    return f"Hello, {name}! Welcome to the world of AI agents."


def run_langgraph_agent(prompt: str = "Please greet Alice and Bob.") -> dict:
    """Run the LangGraph tool-enabled ReAct agent."""
    # ① Load AWS credentials and initialize the recorded tool call trace
    aws_key = os.getenv("AWS_ACCESS_KEY_ID")
    aws_secret = os.getenv("AWS_SECRET_ACCESS_KEY")
    bedrock_model_id = os.getenv("BEDROCK_MODEL_ID", "amazon.nova-lite-v1:0")

    tool_calls_recorded = []

    # ② Attempt LangGraph create_react_agent with Bedrock Converse if AWS keys are configured
    if aws_key and aws_secret:
        try:
            from langchain.chat_models import init_chat_model
            from langchain.tools import tool
            from langgraph.prebuilt import create_react_agent

            @tool
            def greet(name: str) -> str:
                """Greet someone by name."""
                res = greet_tool(name)
                tool_calls_recorded.append(f"greet('{name}') -> {res}")
                return res

            llm = init_chat_model(bedrock_model_id, model_provider="bedrock_converse")
            agent = create_react_agent(model=llm, tools=[greet])
            result = agent.invoke({"messages": [{"role": "user", "content": prompt}]})
            final_message = result["messages"][-1].content
            return {
                "prompt": prompt,
                "response": final_message,
                "tool_calls": tool_calls_recorded,
                "provider": f"LangGraph + Bedrock Converse ({bedrock_model_id})",
            }
        except Exception as exc:
            print(f"Bedrock LangGraph failed ({exc}), falling back to OpenAI")

    # ③ Configure OpenAI client and greet tool schema for the fallback ReAct loop
    client = OpenAI(
        api_key=os.getenv("OPENAI_API_KEY"),
        base_url=os.getenv("OPENAI_BASE_URL", "https://api.openai.com/v1"),
    )
    tools_schema = [
        {
            "type": "function",
            "function": {
                "name": "greet",
                "description": "Greet someone by name.",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "name": {"type": "string", "description": "The name of the person to greet"}
                    },
                    "required": ["name"],
                },
            },
        }
    ]

    messages = [
        {
            "role": "system",
            "content": "You are a friendly LangGraph ReAct agent. Use the greet tool for each person mentioned in the user prompt.",
        },
        {"role": "user", "content": prompt},
    ]

    # ④ Execute the iterative ReAct loop (Reason -> Act -> Observe) up to 5 turns
    for _ in range(5):
        resp = client.chat.completions.create(
            model=os.getenv("OPENAI_MODEL", "gpt-4o-mini"),
            messages=messages,
            tools=tools_schema,
            tool_choice="auto",
        )
        msg = resp.choices[0].message
        messages.append(msg)

        # ⑤ Return final synthesis once the model finishes calling tools
        if not msg.tool_calls:
            return {
                "prompt": prompt,
                "response": msg.content or "",
                "tool_calls": tool_calls_recorded,
                "provider": f"LangGraph ReAct Pattern + OpenAI {os.getenv('OPENAI_MODEL', 'gpt-4o-mini')}",
            }

        # ⑥ Execute each requested greet tool call and feed observations back into messages
        for tc in msg.tool_calls:
            fn_name = tc.function.name
            args = json.loads(tc.function.arguments or "{}")
            if fn_name == "greet":
                output = greet_tool(args.get("name", "Friend"))
                tool_calls_recorded.append(f"greet('{args.get('name')}') -> {output}")
            else:
                output = f"Unknown tool: {fn_name}"
            messages.append({"role": "tool", "tool_call_id": tc.id, "content": output})

    return {
        "prompt": prompt,
        "response": messages[-1].get("content", "Completed greeting process."),
        "tool_calls": tool_calls_recorded,
        "provider": f"LangGraph ReAct Pattern + OpenAI {os.getenv('OPENAI_MODEL', 'gpt-4o-mini')}",
    }


if __name__ == "__main__":
    test_res = run_langgraph_agent("Please greet Alice and Bob.")
    print("Tool calls:", test_res["tool_calls"])
    print("Response:", test_res["response"])

server/main.py

FastAPI route handler for POST /api/langgraph-agent (langgraph_agent_service) in server/main.py.

from dotenv import find_dotenv, load_dotenv
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware

from services import (
    LangGraphAgentRequest,
    LangGraphAgentResponse,
    run_langgraph_agent,
)

# Load environment variables
load_dotenv(find_dotenv())

app = FastAPI(title="AI POCs API")

# Enable CORS for frontend requests
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)


@app.post("/api/langgraph-agent", response_model=LangGraphAgentResponse)
def langgraph_agent_service(req: LangGraphAgentRequest):
    # ① Normalize prompt and invoke LangGraph ReAct greeting agent
    prompt_text = req.prompt.strip() if req.prompt else "Please greet Alice and Bob."
    try:
        res = run_langgraph_agent(prompt_text)
        return LangGraphAgentResponse(**res)
    except Exception as exc:
        raise HTTPException(status_code=500, detail=str(exc))
