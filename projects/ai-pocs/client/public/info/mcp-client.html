How It Works#

The idea behind the demo, the request it sends and the function calls that answer it.

1

Concept#

The Model Context Protocol (MCP) is a standard way for an app to discover and call tools that live in a separate server process. This demo shows the three protocol steps by hand: initialize, tools/list, tools/call.

The server is a separate Python process started by the client and connected through its stdin/stdout. Its data (scores, player stats) is demo data.

2

Theory & Concepts#

  • MCP server: FastMCP turns decorated functions into tools with JSON schemas.
  • stdio transport: JSON-RPC messages over the subprocess's stdin/stdout — which is why the server logs to stderr only.
  • Discovery: the client learns tool names and input schemas at runtime from list_tools().
  • Results: the tool's dict comes back as JSON text in result.content[0].text; unknown tools return isError.

Direct FastMCP Client Service.

"""Direct FastMCP Client Service.

Demonstrates the core Model Context Protocol (MCP) plumbing without an LLM:
1. Starts `cricket_server.py` as a subprocess and talks to it over stdio.
2. Performs the MCP protocol handshake (`session.initialize()`).
3. Lists all registered tools and their JSON schemas from the server (`tools/list`).
4. Invokes an MCP tool directly (`get_live_score` or `get_player_stats`) with structured arguments (`tools/call`).

Run directly:
    python projects/ai-pocs/server/services/29_mcp_client.py
"""

import asyncio
import json
import sys
from pathlib import Path
from typing import Any, Dict, List, Optional

from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import stdio_client
from pydantic import BaseModel, Field


class McpClientRequest(BaseModel):
    """Input payload for Direct MCP Client."""

    tool_name: str = Field(
        default="get_live_score",
        description="Name of tool to call directly: 'get_live_score' or 'get_player_stats'",
    )
    tool_args: Dict[str, Any] = Field(
        default_factory=lambda: {"teams": "India vs West Indies"},
        description="Dictionary of arguments matching tool schema",
    )


class McpToolDescriptor(BaseModel):
    name: str
    description: str
    input_schema: Dict[str, Any]


class McpClientResponse(BaseModel):
    """Output payload from Direct MCP Client."""

    handshake_status: str
    server_tools: List[McpToolDescriptor]
    invoked_tool: str
    arguments: Dict[str, Any]
    result: Dict[str, Any]
    protocol_transport: str


# ① Start cricket_server.py (found next to this file) over stdio using the active Python interpreter
server_script = str(Path(__file__).with_name("cricket_server.py"))
server = StdioServerParameters(command=sys.executable, args=[server_script])


async def run_mcp_client(
    tool_name: str = "get_live_score",
    tool_args: Optional[Dict[str, Any]] = None,
) -> dict:
    """Execute the 3-step MCP client protocol flow (no LLM involved)."""
    tool_args = tool_args or {"teams": "India vs West Indies"}

    # ② Open an MCP stdio transport pipe to the server subprocess
    async with stdio_client(server) as (read, write):
        async with ClientSession(read, write) as session:
            # ③ STEP 1: Perform the MCP protocol handshake
            init = await session.initialize()

            # ④ STEP 2: Discover all available tools and their JSON input schemas
            tools = (await session.list_tools()).tools

            # ⑤ STEP 3: Call one tool directly with structured arguments
            result = await session.call_tool(tool_name, tool_args)

    # ⑥ The server returns the tool's dict as JSON text (or an error message)
    text = result.content[0].text if result.content else "{}"

    return {
        "handshake_status": f"Handshake done with {init.serverInfo.name} (protocol {init.protocolVersion})",
        "server_tools": [
            {"name": t.name, "description": t.description or "", "input_schema": t.inputSchema}
            for t in tools
        ],
        "invoked_tool": tool_name,
        "arguments": tool_args,
        "result": {"error": text} if result.isError else json.loads(text),
        "protocol_transport": "stdio",
    }


if __name__ == "__main__":
    out = asyncio.run(run_mcp_client())
    print(out["handshake_status"])
    print("Tools:", [t["name"] for t in out["server_tools"]])
    print("Result:", json.dumps(out["result"], indent=2))

Cricket FastMCP Server (`cricket_server.py`).

"""Cricket FastMCP Server (`cricket_server.py`).

Provides Model Context Protocol (MCP) tool endpoints for live cricket matches and player statistics:
1. `get_live_score(teams)`: Fetches the live scorecard, target, and match situation.
2. `get_player_stats(player_name)`: Fetches series batting statistics and highest scores.
3. Communicates over standard input/output (`stdio`) transport so MCP clients can spawn it as a subprocess.

Note: The match and player data below is mock/demo data for classroom exercises (no external API key needed).

Run directly (normally the MCP clients 29-31 start it for you as a subprocess):
    python projects/ai-pocs/server/services/cricket_server.py
"""

import sys
from pydantic import Field
from mcp.server.fastmcp import FastMCP

# ---------------------------------------------------------------------------
# FastMCP Server Initialization
# ---------------------------------------------------------------------------
mcp = FastMCP("Cricket-Score-Server")

# ---------------------------------------------------------------------------
# Mock Data Store (Simulated Third-Party Sports API Data)
# ---------------------------------------------------------------------------
LIVE_MATCHES = {
    "india vs west indies": {
        "series": "India vs West Indies ODI Series, Sep-Oct 2026 (DEMO DATA)",
        "match": "2nd ODI",
        "status": "LIVE",
        "india": "287/6 (50 overs)",
        "west_indies": "198/5 (38.2 overs)",
        "target": 288,
        "required": "90 runs from 70 balls",
        "series_score": "India lead 1-0",
    }
}

PLAYER_STATS = {
    "shubman gill": {"team": "India", "series_runs": 164, "innings": 3, "highest": 150},
    "shai hope": {"team": "West Indies", "series_runs": 131, "innings": 2, "highest": 78},
}


# ---------------------------------------------------------------------------
# FastMCP Tool Definitions
# ---------------------------------------------------------------------------
@mcp.tool(description="Get the LIVE score of a cricket match. Use this for any question about the current score, who is winning, or runs required.")
def get_live_score(teams: str = Field(description="The two teams, e.g. 'India vs West Indies'")) -> dict:
    """Get the LIVE score of a cricket match."""
    # ① Log diagnostic trace to stderr (stdout is reserved strictly for MCP JSON-RPC frames)
    print(f"[server] get_live_score called with teams={teams!r}", file=sys.stderr)  # logs go to stderr, NEVER stdout
    # ② Normalize team names and look up the live match scorecard
    key = teams.strip().lower().replace("windies", "west indies")
    if key not in LIVE_MATCHES:
        return {"error": f"No live match found for '{teams}'. Try 'India vs West Indies'."}
    return LIVE_MATCHES[key]


@mcp.tool(description="Get a player's batting stats for the current series. Use this for questions about a specific player's runs.")
def get_player_stats(player_name: str = Field(description="Full player name, e.g. 'Shubman Gill'")) -> dict:
    """Get a player's batting stats for the current series."""
    # ① Log diagnostic trace to stderr
    print(f"[server] get_player_stats called with player_name={player_name!r}", file=sys.stderr)
    # ② Normalize player name and return series batting statistics
    key = player_name.strip().lower()
    if key not in PLAYER_STATS:
        return {"error": f"No stats found for '{player_name}'."}
    return PLAYER_STATS[key]


# ---------------------------------------------------------------------------
# Entry Point (Stdio Transport)
# ---------------------------------------------------------------------------
if __name__ == "__main__":
    mcp.run(transport="stdio")

server/main.py

FastAPI route handler for POST /api/mcp-client (mcp_client_service) in server/main.py.

import asyncio

from dotenv import find_dotenv, load_dotenv
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware

from services import (
    McpClientRequest,
    McpClientResponse,
    run_mcp_client,
)

# Load environment variables
load_dotenv(find_dotenv())

app = FastAPI(title="AI POCs API")

# Enable CORS for frontend requests
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)


@app.post("/api/mcp-client", response_model=McpClientResponse)
def mcp_client_service(req: McpClientRequest):
    # ① Invoke a FastMCP tool directly over stdio, without an LLM
    try:
        res = asyncio.run(run_mcp_client(tool_name=req.tool_name, tool_args=req.tool_args))
        return McpClientResponse(**res)
    except Exception as exc:
        raise HTTPException(status_code=500, detail=str(exc))
