How It Works#

The idea behind the demo, the request it sends and the function calls that answer it.

1

Concept#

🌦️ LangChain Agent replaces manual JSON Schema definitions and tool-call loops with LangChain's @tool decorator and create_agent(model, tools).

By decorating get_weather(city: str) -> str with @tool, LangChain introspects the Python type annotations and docstring to build the OpenAI tool schema automatically. create_agent() then manages the entire Reason → Act → Observe loop until a final answer is reached.

2

Theory & Concepts#

Core Concepts:

  • Schema Introspection via @tool: Function name, type hints, and docstring become the LLM's tool contract without hand-written JSON boilerplate.
  • Automated Agent Loop: create_agent() inspects AIMessage.tool_calls, dispatches the matching Python tool, records a ToolMessage, and re-queries the model automatically.
  • Message Trace Auditing: Iterating over result["messages"] reveals every intermediate tool_calls decision made during the run.
3

Request flow#

🧑 Browser LangchainOneToolCard.tsx sends POST /api/langchain-one-tool with a weather question.
→
FastAPI Route (server/main.py) langchain_one_tool_service(req) validates the message and calls run_langchain_one_tool(message_text).
→
Agent Creation (06_langchain_one_tool.py) create_agent(model=llm, tools=[get_weather]) binds the decorated tool to ChatOpenAI.
→
Agent Execution agent.invoke({"messages": [...]}) calls OpenAI, executes get_weather(city) if requested, and synthesizes the final answer.
→
🧑 Browser Returns (final_answer, tool_called) wrapped in LangchainOneToolResponse.

server/services/06_langchain_one_tool.py

Langchain Version V1 - Agents (single tool)

"""
Langchain Version V1 - Agents (single tool)

Teaching idea: give the agent ONE tool (get_weather), and show how LangChain's
`create_agent` automatically inspects tool signatures, triggers the tool
when needed (e.g. weather queries), and answers general queries directly.

Run directly:
    python projects/ai-pocs/server/services/06_langchain_one_tool.py
"""

import os
from dotenv import find_dotenv, load_dotenv
from langchain.agents import create_agent
from langchain_openai import ChatOpenAI
from pydantic import BaseModel, Field

# ---------------------------------------------------------------------------
# Load environment / API keys
# ---------------------------------------------------------------------------
load_dotenv(find_dotenv(), override=True)

# ---------------------------------------------------------------------------
# Pick your provider here.
# ---------------------------------------------------------------------------
# "groq"   -> free + fast.  Key: https://console.groq.com      (GROQ_API_KEY)
# "gemini" -> free tier.    Key: https://aistudio.google.com   (GEMINI_API_KEY / GOOGLE_API_KEY)
# "openai" -> needs billing/credits set up.                    (OPENAI_API_KEY)
PROVIDER = "groq"

if PROVIDER == "groq":
    MODEL = os.getenv("GROQ_MODEL") or "openai/gpt-oss-20b"
    BASE_URL = os.getenv("GROQ_BASE_URL") or "https://api.groq.com/openai/v1"
    REQUIRED_KEY = os.getenv("GROQ_API_KEY") or "missing-key"
elif PROVIDER == "gemini":
    MODEL = os.getenv("GEMINI_MODEL") or os.getenv("GOOGLE_MODEL") or "gemini-3.8-flash"
    BASE_URL = os.getenv("GEMINI_BASE_URL") or "https://generativelanguage.googleapis.com/v1beta/openai/"
    REQUIRED_KEY = os.getenv("GEMINI_API_KEY") or "missing-key"
else:  # openai
    MODEL = os.getenv("OPENAI_MODEL") or "gpt-4o-mini"
    BASE_URL = os.getenv("OPENAI_BASE_URL") or "https://api.openai.com/v1"
    REQUIRED_KEY = os.getenv("OPENAI_API_KEY") or "missing-key"


# ---------------------------------------------------------------------------
# Request & Response Schemas (Pydantic)
# ---------------------------------------------------------------------------
class LangchainOneToolRequest(BaseModel):
    """Input payload for single-tool LangChain agent queries."""

    message: str = Field(
        default="What is the weather like in New York?",
        description="Question for the single-tool LangChain agent",
    )


class LangchainOneToolResponse(BaseModel):
    """Output payload containing the final response and tool call indicator."""

    message: str
    response: str
    tool_called: bool = False


# ---------------------------------------------------------------------------
# Tools
# ---------------------------------------------------------------------------
def get_weather(city: str) -> str:
    """Get the current weather for a given city."""
    return f"The weather in {city} is too hot to handle, 28 degrees C."


TOOLS = [get_weather]

# ---------------------------------------------------------------------------
# Build the agent
# ---------------------------------------------------------------------------
# ① Initialize the LLM client pointed at the configured provider
llm = ChatOpenAI(
    model=MODEL,
    api_key=REQUIRED_KEY,
    base_url=BASE_URL if BASE_URL else None,
    temperature=0,
    timeout=20.0,
)

# ② Create the LangChain tool-calling agent with get_weather and system instructions
agent = create_agent(
    model=llm,
    tools=TOOLS,
    system_prompt=(
        "You are a helpful assistant. Use the available tools when needed. "
        "Always state the final result clearly and completely in your answer."
    ),
)


# ---------------------------------------------------------------------------
# Core Service Logic
# ---------------------------------------------------------------------------
def run_langchain_one_tool(
    question: str = "What is the weather like in New York?",
) -> tuple[str, bool]:
    """Run a user question through the single-tool LangChain agent."""
    # ① Invoke the LangChain agent with the user's question
    response = agent.invoke({"messages": [{"role": "user", "content": question}]})
    messages = response["messages"]
    # ② Check whether any tool message appears in the trace
    tool_called = any(msg.type == "tool" for msg in messages)
    # ③ Extract and return the final assistant reply alongside the tool flag
    final_msg = messages[-1]
    answer = str(final_msg.content) if hasattr(final_msg, "content") else str(final_msg)
    return answer, tool_called


def ask(question: str) -> None:
    answer, tool_called = run_langchain_one_tool(question)
    print(f"\nQ: {question}")
    print(f"Tool called: {'Yes' if tool_called else 'No'}")
    print(f"A: {answer}")


# ---------------------------------------------------------------------------
# Demo
# ---------------------------------------------------------------------------
if __name__ == "__main__":
    ask("What is the weather like in New York?")
    ask("What is scaler company?")

FastAPI route handler for POST /api/langchain-one-tool (langchain_one_tool_service) in server/main.py.

from dotenv import find_dotenv, load_dotenv
from fastapi import FastAPI, HTTPException
from fastapi.middleware.cors import CORSMiddleware

from services import (
    LangchainOneToolRequest,
    LangchainOneToolResponse,
    run_langchain_one_tool,
)

# Load environment variables
load_dotenv(find_dotenv())

app = FastAPI(title="AI POCs API")

# Enable CORS for frontend requests
app.add_middleware(
    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)


@app.post("/api/langchain-one-tool", response_model=LangchainOneToolResponse)
@app.post("/api/langchain-agent", response_model=LangchainOneToolResponse)
def langchain_one_tool_service(req: LangchainOneToolRequest):
    # ① Validate message and invoke LangChain single-tool agent
    message_text = req.message.strip() if req.message else "What is the weather like in New York?"
    if not message_text:
        raise HTTPException(status_code=400, detail="Message cannot be empty")
    try:
        reply, tool_called = run_langchain_one_tool(message_text)
        return LangchainOneToolResponse(
            message=message_text,
            response=reply,
            tool_called=tool_called,
        )
    except Exception as exc:
        raise HTTPException(status_code=500, detail=str(exc))
