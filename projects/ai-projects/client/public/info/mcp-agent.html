Concept#

🏏 FastMCP Cricket Agent bridges an LLM (Groq Llama 3.3 70B or OpenAI GPT-4o-mini) to the FastMCP Cricket Server so natural-language questions automatically route to get_live_score or get_player_stats.

While Demo 29 called MCP tools manually, this service exposes the FastMCP server's tool schemas to the LLM. When a user asks "What is the score in India vs West Indies?" or "How many runs has Shubman Gill scored?", the LLM selects the right MCP tool, extracts the arguments, executes the call against cricket_server.py, and turns the live JSON into a grounded commentary response.

2

Theory & Concepts#

Core Concepts:

  • MCP-to-Function-Calling Bridge: Translates FastMCP tool descriptors into the OpenAI/Groq tools=[{"type": "function", ...}] schema format.
  • Dynamic Argument Extraction: The LLM extracts entity names (e.g., "India vs West Indies" or "Shubman Gill") from freeform user questions.
3

Request flow#

🧑 Browser McpAgentCard.tsx sends POST /api/mcp-agent with a cricket question.
→
FastAPI Route (server/main.py) mcp_agent_service(req) calls run_mcp_agent(req.question).
→
Tool Decision (30_mcp_agent.py) Sends the user question and FastMCP tool schemas to Groq/OpenAI.
→
FastMCP Execution (cricket_server.py) Dispatches get_live_score or get_player_stats with the LLM-extracted arguments.
→
Grounded Answer Feeds the MCP JSON payload back to the LLM to generate the final answer.
→
🧑 Browser Returns McpAgentResponse with the final response, tool_called, tool_arguments, and tool_result.

MCP-Enabled LLM Agent Service.

"""MCP-Enabled LLM Agent Service.

Demonstrates combining LLM intelligence with Model Context Protocol (MCP) tools:
1. Translates FastMCP tool definitions into LLM function calling schemas.
2. The LLM decides WHICH tool to invoke (`get_live_score`, `get_player_stats`) and extracts arguments.
3. Our code executes the tool against the MCP cricket server and feeds the output back.
4. Generates an accurate, factually grounded answer.

Run directly:
    python projects/ai-projects/server/services/30_mcp_agent.py
"""

import json
import os
from typing import Any, Dict, List, Optional
from dotenv import find_dotenv, load_dotenv
from openai import OpenAI
from pydantic import BaseModel, Field

from services.mcp.cricket_server import get_live_score, get_player_stats

load_dotenv(find_dotenv())


class McpAgentRequest(BaseModel):
    """Input payload for MCP Agent."""

    question: str = Field(
        default="What is the score in India vs West Indies?",
        description="Natural language question about cricket scores or player stats",
    )


class McpAgentResponse(BaseModel):
    """Output payload from MCP Agent."""

    question: str
    response: str
    tool_called: Optional[str]
    tool_arguments: Optional[Dict[str, Any]]
    tool_result: Optional[Dict[str, Any]]
    provider: str


def run_mcp_agent(question: str = "What is the score in India vs West Indies?") -> dict:
    """Run the MCP-Enabled LLM agent."""
    # ① Select Groq or OpenAI client based on available environment credentials
    groq_key = os.getenv("GROQ_API_KEY")
    openai_key = os.getenv("OPENAI_API_KEY")

    if groq_key:
        client = OpenAI(api_key=groq_key, base_url=os.getenv("GROQ_BASE_URL", "https://api.groq.com/openai/v1"))
        model = os.getenv("GROQ_MODEL", "llama-3.3-70b-versatile")
        provider_name = f"Groq ({model}) + FastMCP"
    else:
        client = OpenAI(api_key=openai_key, base_url=os.getenv("OPENAI_BASE_URL", "https://api.openai.com/v1"))
        model = os.getenv("OPENAI_MODEL", "gpt-4o-mini")
        provider_name = f"OpenAI ({model}) + FastMCP"

    # ② Convert FastMCP server tool definitions into LLM function-calling schemas
    tools_schema = [
        {
            "type": "function",
            "function": {
                "name": "get_live_score",
                "description": "Get the LIVE score of a cricket match. Use this for questions about the current score, who is winning, or runs required.",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "teams": {"type": "string", "description": "The two teams playing, e.g. 'India vs West Indies'"}
                    },
                    "required": ["teams"],
                },
            },
        },
        {
            "type": "function",
            "function": {
                "name": "get_player_stats",
                "description": "Get a player's batting stats for the current series. Use this for questions about a specific player's runs or innings.",
                "parameters": {
                    "type": "object",
                    "properties": {
                        "player_name": {"type": "string", "description": "Full player name, e.g. 'Shubman Gill'"}
                    },
                    "required": ["player_name"],
                },
            },
        },
    ]

    # ③ Send user question and MCP tool schemas to the model
    messages = [
        {
            "role": "system",
            "content": "You are a cricket assistant with access to real-time match data via FastMCP tools. Use tools when asked about scores or player stats.",
        },
        {"role": "user", "content": question},
    ]

    resp = client.chat.completions.create(
        model=model,
        messages=messages,
        tools=tools_schema,
        tool_choice="auto",
    )
    msg = resp.choices[0].message

    if not msg.tool_calls:
        return {
            "question": question,
            "response": msg.content or "",
            "tool_called": None,
            "tool_arguments": None,
            "tool_result": None,
            "provider": provider_name,
        }

    # ④ Execute the selected tool against the FastMCP cricket server
    tc = msg.tool_calls[0]
    fname = tc.function.name
    args = json.loads(tc.function.arguments or "{}")

    if fname == "get_live_score":
        tool_res = get_live_score(args.get("teams", "India vs West Indies"))
    elif fname == "get_player_stats":
        tool_res = get_player_stats(args.get("player_name", "Shubman Gill"))
    else:
        tool_res = {"error": f"Unknown tool: {fname}"}

    # ⑤ Feed the MCP tool output back to the LLM to synthesize the final grounded answer
    messages.append(msg)
    messages.append({"role": "tool", "tool_call_id": tc.id, "content": json.dumps(tool_res)})

    final_resp = client.chat.completions.create(model=model, messages=messages)
    final_text = final_resp.choices[0].message.content or ""

    return {
        "question": question,
        "response": final_text,
        "tool_called": fname,
        "tool_arguments": args,
        "tool_result": tool_res,
        "provider": provider_name,
    }


if __name__ == "__main__":
    out = run_mcp_agent()
    print("Tool Called:", out["tool_called"])
    print("Response:", out["response"])

Cricket MCP Server.

"""Cricket MCP Server.

Provides Model Context Protocol (MCP) tool endpoints for cricket matches and player statistics:
1. `get_live_score(teams)`: Fetches live scorecard and match situation.
2. `get_player_stats(player_name)`: Fetches series stats and high scores.

Run directly:
    python projects/ai-projects/server/services/mcp/cricket_server.py
"""

import sys
from pydantic import Field

LIVE_MATCHES = {
    "india vs west indies": {
        "series": "India vs West Indies ODI Series, Sep-Oct 2026 (DEMO DATA)",
        "match": "2nd ODI",
        "status": "LIVE",
        "india": "287/6 (50 overs)",
        "west_indies": "198/5 (38.2 overs)",
        "target": 288,
        "required": "90 runs from 70 balls",
        "series_score": "India lead 1-0",
    }
}

PLAYER_STATS = {
    "shubman gill": {"team": "India", "series_runs": 164, "innings": 3, "highest": 150},
    "shai hope": {"team": "West Indies", "series_runs": 131, "innings": 2, "highest": 78},
}


def get_live_score(teams: str) -> dict:
    """Get the LIVE score of a cricket match."""
    # ① Normalize team names and look up the live match scorecard
    key = teams.strip().lower().replace("windies", "west indies")
    if key not in LIVE_MATCHES:
        return {"error": f"No live match found for '{teams}'. Try 'India vs West Indies'."}
    return LIVE_MATCHES[key]


def get_player_stats(player_name: str) -> dict:
    """Get a player's batting stats for the current series."""
    # ① Normalize player name and return series batting statistics
    key = player_name.strip().lower()
    if key not in PLAYER_STATS:
        return {"error": f"No stats found for '{player_name}'."}
    return PLAYER_STATS[key]


# If FastMCP is available, register tools
try:
    from mcp.server.fastmcp import FastMCP

    mcp = FastMCP("Cricket-Score-Server")

    @mcp.tool(description="Get the LIVE score of a cricket match. Use this for questions about the current score or runs required.")
    def mcp_get_live_score(teams: str = Field(description="The two teams, e.g. 'India vs West Indies'")) -> dict:
        return get_live_score(teams)

    @mcp.tool(description="Get a player's batting stats for the current series. Use this for questions about a specific player's runs.")
    def mcp_get_player_stats(player_name: str = Field(description="Full player name, e.g. 'Shubman Gill'")) -> dict:
        return get_player_stats(player_name)

    if __name__ == "__main__":
        mcp.run(transport="stdio")
except ImportError:
    if __name__ == "__main__":
        print("FastMCP not installed, direct functions available.")
